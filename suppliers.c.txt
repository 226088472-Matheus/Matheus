#include <stdio.h>
#include <string.h>
#include "suppliers.h"

int addSupplier(struct Supplier list[], int count) {
    if (count < 10) {
        printf("Enter name: ");
        scanf("%s", list[count].name);
        printf("Enter email: ");
        scanf("%s", list[count].email);
        printf("Enter phone: ");
        scanf("%s", list[count].phone);
        printf("Enter town: ");
        scanf("%s", list[count].town);
        count++;
        printf("Supplier added successfully!\n");
    } else {
        printf("Supplier list is full!\n");
    }
    return count;
}

void displaySuppliers(struct Supplier list[], int count) {
    int i;
    for (i = 0; i < count; i++) {
        printf("\nSupplier %d:\n", i + 1);
        printf("Name: %s\n", list[i].name);
        printf("Town: %s\n", list[i].town);
    }
}

void searchSupplier(struct Supplier list[], int count) {
    char searchName[50];
    int i, found = 0;
    printf("Enter name to search: ");
    scanf("%s", searchName);
    for (i = 0; i < count; i++) {
        if (strcmp(list[i].name, searchName) == 0) {
            printf("Supplier found in %s!\n", list[i].town);
            found = 1;
        }
    }
    if (!found) {
        printf("Supplier not found.\n");
    }
}
